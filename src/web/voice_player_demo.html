工程调用方式 (SDK 接入模式)

// 1. 引入客户端与伴读控制器 (前端零密钥，直连同源本地网关)

import { NextdoorVoiceClient, ArticleAudioPlayer } from '@/gateway/client';


// 2. 初始化客户端

const client = new NextdoorVoiceClient({ baseURL: 'http://127.0.0.1:8090' });


// 3. 创建伴读播放器并载入长文章 (自动切片 ≤ 280 字符并静默预加载)

const player = new ArticleAudioPlayer({

  client,

  defaultVoiceAlias: 'standard_female_warm',

  onChunkStart: (index, total, text) => {

    console.log(`正在播报第 ${index + 1}/${total} 段: ${text}`);

  }

});

player.loadArticle(document.querySelector('article').innerText);

player.play();
